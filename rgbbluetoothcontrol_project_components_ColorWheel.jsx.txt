// ColorWheel — interactive hue-saturation wheel.
// Renders a conic-gradient hue ring with a white-to-transparent radial
// overlay for saturation. Clicking/dragging picks a color; disabled state
// applies a grayscale + lock overlay.

function hsvToRgb(h, s, v) {
  const c = v * s;
  const hh = (h / 60) % 6;
  const x = c * (1 - Math.abs((hh % 2) - 1));
  let r = 0, g = 0, b = 0;
  if (hh < 1) { r = c; g = x; }
  else if (hh < 2) { r = x; g = c; }
  else if (hh < 3) { g = c; b = x; }
  else if (hh < 4) { g = x; b = c; }
  else if (hh < 5) { r = x; b = c; }
  else { r = c; b = x; }
  const m = v - c;
  return [
    Math.round((r + m) * 255),
    Math.round((g + m) * 255),
    Math.round((b + m) * 255),
  ];
}

function rgbToHex(r, g, b) {
  return '#' + [r, g, b].map(v => v.toString(16).padStart(2, '0')).join('').toUpperCase();
}

function ColorWheel({
  size = 280,
  enabled = true,
  color = { h: 0, s: 0, v: 1 },
  onChange,
  ringWidth = null, // null = filled disc
  accent = '#FFFFFF',
}) {
  const ref = React.useRef(null);
  const [dragging, setDragging] = React.useState(false);

  const [rgbR, rgbG, rgbB] = hsvToRgb(color.h, color.s, color.v);

  // Marker position (in px from center)
  const radius = size / 2;
  const markerR = color.s * (radius - 14);
  const markerAngle = (color.h - 90) * Math.PI / 180;
  const markerX = radius + markerR * Math.cos(markerAngle);
  const markerY = radius + markerR * Math.sin(markerAngle);

  function pick(e) {
    if (!enabled) return;
    const rect = ref.current.getBoundingClientRect();
    const cx = rect.width / 2;
    const cy = rect.height / 2;
    const x = (e.clientX ?? e.touches?.[0]?.clientX) - rect.left - cx;
    const y = (e.clientY ?? e.touches?.[0]?.clientY) - rect.top - cy;
    const dist = Math.sqrt(x * x + y * y);
    const maxR = rect.width / 2 - 2;
    const s = Math.min(1, dist / maxR);
    let h = Math.atan2(y, x) * 180 / Math.PI + 90;
    if (h < 0) h += 360;
    onChange?.({ h, s, v: color.v });
  }

  React.useEffect(() => {
    if (!dragging) return;
    const move = (e) => pick(e);
    const up = () => setDragging(false);
    window.addEventListener('pointermove', move);
    window.addEventListener('pointerup', up);
    return () => {
      window.removeEventListener('pointermove', move);
      window.removeEventListener('pointerup', up);
    };
  }, [dragging, color.v]);

  const hueGradient = 'conic-gradient(from 0deg, red, yellow, lime, cyan, blue, magenta, red)';
  const satGradient = 'radial-gradient(circle, white 0%, rgba(255,255,255,0) 70%)';

  return (
    <div
      ref={ref}
      onPointerDown={(e) => { if (enabled) { setDragging(true); pick(e); } }}
      style={{
        width: size,
        height: size,
        borderRadius: '50%',
        position: 'relative',
        background: hueGradient,
        cursor: enabled ? (dragging ? 'grabbing' : 'crosshair') : 'not-allowed',
        filter: enabled ? 'none' : 'grayscale(1) brightness(0.85)',
        opacity: enabled ? 1 : 0.5,
        transition: 'filter .25s, opacity .25s',
        touchAction: 'none',
        boxShadow: enabled
          ? '0 8px 32px rgba(0,0,0,0.18), inset 0 0 0 1px rgba(255,255,255,0.1)'
          : 'inset 0 0 0 1px rgba(0,0,0,0.08)',
      }}
    >
      {/* Saturation overlay (white center fading out) */}
      <div style={{
        position: 'absolute', inset: 0, borderRadius: '50%',
        background: satGradient,
        pointerEvents: 'none',
      }} />

      {/* Optional ring cutout (donut style) */}
      {ringWidth && (
        <div style={{
          position: 'absolute',
          inset: ringWidth,
          borderRadius: '50%',
          background: enabled ? `rgb(${rgbR},${rgbG},${rgbB})` : '#bbb',
          boxShadow: 'inset 0 2px 12px rgba(0,0,0,0.15)',
          transition: 'background .2s',
          pointerEvents: 'none',
        }} />
      )}

      {/* Picker marker */}
      {enabled && !ringWidth && (
        <div style={{
          position: 'absolute',
          left: markerX - 14,
          top: markerY - 14,
          width: 28, height: 28,
          borderRadius: '50%',
          border: '3px solid white',
          boxShadow: '0 2px 8px rgba(0,0,0,0.35), inset 0 0 0 1px rgba(0,0,0,0.2)',
          background: `rgb(${rgbR},${rgbG},${rgbB})`,
          pointerEvents: 'none',
        }} />
      )}

      {/* Lock overlay when disabled */}
      {!enabled && (
        <div style={{
          position: 'absolute', inset: 0,
          display: 'flex', alignItems: 'center', justifyContent: 'center',
          flexDirection: 'column', gap: 8,
          color: 'rgba(0,0,0,0.55)',
          fontFamily: 'Roboto, system-ui, sans-serif',
          pointerEvents: 'none',
        }}>
          <svg width="36" height="36" viewBox="0 0 24 24" fill="none">
            <rect x="5" y="11" width="14" height="9" rx="2" fill="currentColor"/>
            <path d="M8 11V7a4 4 0 018 0v4" stroke="currentColor" strokeWidth="2" fill="none"/>
          </svg>
          <div style={{ fontSize: 13, fontWeight: 500, letterSpacing: 0.3 }}>
            Connect a device
          </div>
        </div>
      )}
    </div>
  );
}

Object.assign(window, { ColorWheel, hsvToRgb, rgbToHex });
